import React, { useState } from "react";

function ProfileCard({ name, bio, profilePicture }) {
  const [color, setColor] = useState("#f0f0f0");

  return (
    <div
      style={{ backgroundColor: color }}
      onMouseEnter={() => setColor("#d1c4e9")}
      onMouseLeave={() => setColor("#f0f0f0")}
    >
      <img src={profilePicture} alt={name} width="150" />

      <h2>{name}</h2>
      <p>{bio}</p>
    </div>
  );
}

export default ProfileCard;
